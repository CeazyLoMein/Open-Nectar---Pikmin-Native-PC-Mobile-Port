cmake_minimum_required(VERSION 3.24)
project(NectarCoopAppProbe LANGUAGES C CXX OBJCXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" OR NOT CMAKE_OSX_SYSROOT MATCHES "[Ii][Pp]hone[Ss]imulator")
    message(FATAL_ERROR "This asset-free test app requires iOS Simulator")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(NECTAR_NET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NECTAR_NET_ENABLE_GNS ON CACHE BOOL "" FORCE)
set(NECTAR_NET_ENABLE_UPNP OFF CACHE BOOL "" FORCE)
set(NECTAR_NET_BUILD_COOP_SOCKET_PROBE OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/libs/nectar_net" "${CMAKE_CURRENT_BINARY_DIR}/nectar-core")
# Compile the EXACT same test body; only rename its entry point.
add_library(socket_body OBJECT "${ROOT}/libs/nectar_net/tests/coop_socket_probe.cpp")
target_compile_definitions(socket_body PRIVATE main=nectar_coop_socket_probe_main)
target_link_libraries(socket_body PRIVATE nectar_net)
target_compile_features(socket_body PRIVATE cxx_std_17)
target_compile_options(socket_body PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(health_body OBJECT "${ROOT}/tests/ios_simulator_health.c")
target_compile_definitions(health_body PRIVATE main=nectar_simulator_health_main)
target_compile_options(health_body PRIVATE -Wall -Wextra -Werror)
foreach(KIND IN ITEMS socket health)
    if(KIND STREQUAL "socket")
        set(PROBE_TARGET NectarCoopSocketProbe)
        set(PROBE_ID org.opennectar.tests.coopsocket)
        set(HEALTH 0)
    else()
        set(PROBE_TARGET NectarCoopHealthProbe)
        set(PROBE_ID org.opennectar.tests.coophealth)
        set(HEALTH 1)
    endif()
    add_executable(${PROBE_TARGET} MACOSX_BUNDLE main.mm $<TARGET_OBJECTS:${KIND}_body>)
    target_compile_features(${PROBE_TARGET} PRIVATE cxx_std_17)
    target_compile_definitions(${PROBE_TARGET} PRIVATE NECTAR_PROBE_HEALTH=${HEALTH})
    target_compile_options(${PROBE_TARGET} PRIVATE -fobjc-arc -Wall -Wextra -Werror)
    target_link_libraries(${PROBE_TARGET} PRIVATE "-framework UIKit" "-framework Foundation")
    if(KIND STREQUAL "socket")
        target_link_libraries(${PROBE_TARGET} PRIVATE nectar_net)
    endif()
    # Health app has its own link: no nectar_net/GNS/game initialization.
    configure_file(Info.plist.in "${CMAKE_CURRENT_BINARY_DIR}/${PROBE_TARGET}-Info.plist" @ONLY)
    set_target_properties(${PROBE_TARGET} PROPERTIES
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/${PROBE_TARGET}-Info.plist"
        XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${PROBE_ID}"
        XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1"
        XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED "NO")
endforeach()
